function initJitsi() {
if(jitsiApi) return; // Ya existe una instancia
document.getElementById('jitsi-placeholder').classList.add('hidden');
// Opciones de la API de Jitsi

const options = {
roomName: 'ConsultorioSereno_Privado_' + Math.random().toString(36).substring(7),
width: '100%',
height: '100%',
parentNode: document.getElementById('jitsi-container'),
lang: 'es',

displayName: isUserLoggedIn ? currentUser.name : 'Paciente Invitado'
},
configOverwrite: {
prejoinPageEnabled: false, // Entrar directo sin sala previa
startWithAudioMuted: true
},
interfaceConfigOverwrite: {

'microphone', 'camera', 'closedcaptions', 'desktop', 'fullscreen',
'fodeviceselection', 'hangup', 'profile', 'chat', 'settings', 'videoquality'
],
}
};
jitsiApi = new JitsiMeetExternalAPI(domain, options);
}
function disposeJitsi() {
